Số nguyên tố đối xứng

Kỹ thuật cốt lõi: sinh trực tiếp số đối xứng từ nửa trái (không duyệt qua toàn bộ N), rồi kiểm tra nguyên tố.

Đề bài

Một số được gọi là số nguyên tố đối xứng nếu nó vừa là số nguyên tố, vừa đọc xuôi đọc ngược giống nhau (ví dụ 131). Cho số nguyên dương N, đếm số lượng số nguyên tố đối xứng lớn hơn 10 và bé hơn N.

Ví dụ: N = 150 → đáp án = 3 (gồm 11, 101, 131).
12 ┃ (gương) 21 Chỉ cần biết nửa trái → suy ra toàn bộ số đối xứng

#include <bits/stdc++.h>
using namespace std;
#define ll long long

bool ktnt(ll a) { // kiểm tra số nguyên tố
    if (a < 2) return 0;
    if (a == 2 || a == 3) return 1;
    if (a % 2 == 0 || a % 3 == 0) return 0;
    for (int j = 5; j <= sqrt(a); j += 6) {
        if (a % j == 0 || a % (j + 2) == 0) return 0;
    }
    return 1;
}

ll dx1(ll x) { // ghép dạng LẺ: NkN
    ll y = x / 10;
    while (y > 0) {
        x = x * 10 + (y % 10);
        y = y / 10;
    }
    return x;
}

ll dx2(ll x) { // ghép dạng CHẴN: NkkN
    ll y = x;
    while (y > 0) {
        x = x * 10 + (y % 10);
        y = y / 10;
    }
    return x;
}

int main() {
    ll n, res = 0;
    cin >> n;
    for (ll i = 1; i < 100000; i++) {
        ll k = dx1(i);
        if (k >= n) break;
        if (k > 10 && ktnt(k)) res++;
        k = dx2(i);
        if (k < n && ktnt(k)) res++;
    }
    cout << res;
}
import math

def ktnt(a):  # kiểm tra số nguyên tố
    if a < 2:
        return False
    if a in (2, 3):
        return True
    if a % 2 == 0 or a % 3 == 0:
        return False
    j = 5
    while j * j <= a:
        if a % j == 0 or a % (j + 2) == 0:
            return False
        j += 6
    return True

def dx1(x):  # ghép dạng LẺ: NkN
    y = x // 10
    while y > 0:
        x = x * 10 + (y % 10)
        y = y // 10
    return x

def dx2(x):  # ghép dạng CHẴN: NkkN
    y = x
    while y > 0:
        x = x * 10 + (y % 10)
        y = y // 10
    return x

def giai(n):
    res = 0
    i = 1
    while i < 100000:
        k = dx1(i)
        if k >= n:
            break
        if k > 10 and ktnt(k):
            res += 1
        k = dx2(i)
        if k < n and ktnt(k):
            res += 1
        i += 1
    return res

if __name__ == "__main__":
    n = int(input())
    print(giai(n))
